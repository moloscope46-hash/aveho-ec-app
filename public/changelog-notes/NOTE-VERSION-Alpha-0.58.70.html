✅ Fix appliqué

Nouveau helper lib/chambres.js qui (a) sonde la présence de la colonne UNE seule fois, (b) cache le résultat module-level pour éviter toute boucle, (c) fallback gracieux si absente.

// lib/chambres.js — 0.58.70
let _hasBatimentCol = null;  // ← cache module-level, jamais re-sondé après 1er appel

export async function chambresHasBatimentCol(supabase) {
  if (_hasBatimentCol !== null) return _hasBatimentCol;  // ← anti-boucle
  const { error } = await supabase.from('chambres')
    .select('batiment_id', { count: 'exact', head: true }).limit(1);
  if (error && (error.code === '42703' || /batiment_id/i.test(error.message))) {
    _hasBatimentCol = false;
    return false;
  }
  _hasBatimentCol = true;
  return true;
}

export async function selectChambresContexte(supabase, { serviceId, batimentId }) {
  const hasCol = await chambresHasBatimentCol(supabase);
  if (serviceId) {
    const cols = hasCol ? 'id, service_id, batiment_id' : 'id, service_id';
    const r = await supabase.from('chambres').select(cols).eq('service_id', serviceId);
    return { data: r.data || [], hasBatimentCol: hasCol };
  }
  if (batimentId) {
    if (!hasCol) return { data: [], hasBatimentCol: false };  // ← fallback propre
    const r = await supabase.from('chambres').select('id, service_id, batiment_id').eq('batiment_id', batimentId);
    return { data: r.data || [], hasBatimentCol: true };
  }
}
