Toutes ces fonctions avaient CORS '*' et AUCUNE vérification d'authentification. Le commentaire dans le code disait "appeler avec Bearer SERVICE_ROLE_KEY" mais aucun check côté serveur ! Un attaquant connaissant l'URL pouvait les déclencher à volonté. CVSS estimé : 7.5 HIGH.

✅ Le fix : header x-cron-secret + helper requireCronSecret

Nouveau helper dans _shared/auth.ts

export function requireCronSecret(req: Request): Response | null {
  const expected = Deno.env.get("CRON_SECRET");

  if (!expected) {
    // Fail-secure : refuse si pas de secret configuré
    return new Response(
      JSON.stringify({ error: "CRON_SECRET non configuré" }),
      { status: 500, headers: { ... } }
    );
  }

  const provided = req.headers.get("x-cron-secret") || "";

  // Comparaison constant-time (anti timing-attacks)
  if (provided.length !== expected.length) return new Response(..., { status: 401 });
  let mismatch = 0;
  for (let i = 0; i < provided.length; i++) {
    mismatch |= provided.charCodeAt(i) ^ expected.charCodeAt(i);
  }
  if (mismatch !== 0) return new Response(..., { status: 401 });

  return null;  // OK
}

Usage dans chaque CRON Function

import { buildCorsHeaders, requireCronSecret } from "../_shared/auth.ts";

Deno.serve(async (req) => {

🔒 16e LINT anti-régression critique

it("Toutes les CRON Functions importent requireCronSecret", () => {
  const CRON_FUNCTIONS_REQUIRING_SECRET = [
    "auto-archive-consents", "maintenance-daily-cron",
    "send-digest", "send-renouvellement-rappels", "weekly-stats-digest",
  ];
  const violations = [];
  for (const fn of CRON_FUNCTIONS_REQUIRING_SECRET) {
    const src = fs.readFileSync(`supabase/functions/${fn}/index.ts`, "utf-8");
    if (!/requireCronSecret/.test(src)) violations.push(fn);
  }
  expect(violations).toEqual([]);  // ← FAIL si nouvelle CRON sans secret
});
