// 0.58.54 - lib/useContextPatientIds.js (NEW)
export function useContextPatientIds() {
  const ctx = useCurrentContext();
  const [patientIds, setPatientIds] = useState(null);
  
  useEffect(() => {
    if (!ctx.active || (!ctx.batimentId && !ctx.serviceId)) {
      setPatientIds(null);  // pas de filtrage
      return;
    }
    (async () => {
      // 1) chambres dans le périmètre
      let query = supabase.from('chambres').select('id, service_id, batiment_id');
      if (ctx.serviceId) query = query.eq('service_id', ctx.serviceId);
      else query = query.eq('batiment_id', ctx.batimentId);
      const { data: chambres } = await query;
      const chambreIds = chambres.map(c => c.id);
      
      // 2) patients liés à ces chambres
      const { data: pats } = await supabase.from('patients')
        .select('id').in('chambre_id', chambreIds);
      
      setPatientIds(new Set(pats.map(p => p.id)));
    })();
  }, [ctx.active, ctx.batimentId, ctx.serviceId]);
  
  return { patientIds, ctx };
}

// Usage dans /commandes /achats /signalements
const { patientIds, ctx } = useContextPatientIds();

const filtered = ctx.active && patientIds
  ? rows.filter(r => !r.patient_id || patientIds.has(r.patient_id))
  : rows;

💡 Note : les rows sans patient_id (commandes au niveau étab) sont conservées. Seules les rows liées à un patient hors périmètre sont masquées. Empty state contextuel : "Aucune commande dans ce périmètre" si filtre actif vs message générique sinon.

💊 2. Ajout partenaire depuis FINESS

Sur /partenaires-rpps, en complément du bouton "Ajouter depuis RPPS" (médecins, IDE individuels), un nouveau bouton vert "Ajouter depuis FINESS" permet d'ajouter des pharmacies d'officine, des SSIAD (Services de Soins Infirmiers À Domicile), des services HAD, ou tout autre établissement de santé.

🩺 Ajouter depuis RPPS (médecins, IDE)
🏥 Ajouter depuis FINESS (pharmacie, SSIAD)

Catégories FINESS pré-filtrées

  • Pharmacies & Matériel médical (LPP) :
    • Pharmacies d'Officine (code 620)
    • Pharmacies à Usage Intérieur (code 619)
    • Commerces Biens à Usage Médical / LPP (code 3201)
    • Loueurs Matériel Médical (code 3299)
  • Soins à domicile :
    • HAD - Hospitalisation à Domicile (code 354)
    • SSIAD - Services de Soins Infirmiers À Domicile
    • SPASAD (code 446)

Les filtres sont modifiables directement dans la recherche pour étendre à d'autres catégories.

// 0.58.54 - app/partenaires-rpps/page.js
async function addPartenaireFromFiness(f) {
  const typeFiness = (f.type || '').toLowerCase();
  const isPharmacie = typeFiness.includes('pharmac');
  const isSsiad = typeFiness.match(/ssiad|infirm|had/i);
  
  const payload = {
    structure_id: auth.structureId,
    rpps: f.finess,  // stocke le FINESS dans le champ rpps
    nom: f.nom,
    profession: isPharmacie ? 'Pharmacie d\'officine'
              : isSsiad ? 'SSIAD / Infirmières'
              : f.type,
    adresse: f.adresse, cp: f.code_postal, commune: f.ville,
    telephone: f.telephone,
    est_pharmacien: isPharmacie,    // détection automatique
    est_intervenant: isSsiad,
    est_prescripteur: false,
  };
  
  await supabase.from('partenaires_rpps').insert(payload);
}
