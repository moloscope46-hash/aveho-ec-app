🔍 Filtrage patients par contexte bât/svc

Le BatimentServiceSwitcher de la TopBar (ajouté en 0.58.35) émet désormais des events av-current-context-change que /patients écoute. Un nouveau bouton "Filtrer par contexte" apparaît à côté de "Filtres avancés" quand un contexte est défini → filtre la liste pour ne montrer que les patients du bâtiment/service courant.

// 0.58.38 - app/patients/page.js
const [ctxFilter, setCtxFilter] = useState({ batimentId: null, serviceId: null, active: false });

useEffect(() => {
  function onCtxChange(e) {
    const detail = e?.detail || {};
    setCtxFilter(prev => ({ ...prev, batimentId: detail.batimentId, serviceId: detail.serviceId }));
  }
  window.addEventListener('av-current-context-change', onCtxChange);
  return () => window.removeEventListener('av-current-context-change', onCtxChange);
}, []);

// Filtrage enrichi
const filtered = rows.filter((r) => {
  // ... filtres existants
  if (ctxFilter.active && (ctxFilter.batimentId || ctxFilter.serviceId)) {
    const ch = chambres.find((c) => c.id === r.chambre_id);
    if (!ch) return false;
    if (ctxFilter.serviceId && ch.service_id !== ctxFilter.serviceId) return false;
    if (ctxFilter.batimentId && ch.batiment_id !== ctxFilter.batimentId) return false;
  }
  return true;
});
