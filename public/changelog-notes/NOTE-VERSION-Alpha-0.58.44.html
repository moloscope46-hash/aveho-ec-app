🩹 1. Fix CRITIQUE : dialogs.prompt is not a function

Bug prod confirmé : tous les composants utilisant dialogs.prompt plantaient en prod minifiée (L.dialogs.prompt is not a function). Concernés : LiensFavoris (édit/ajout liens), Notes (addTab/renameTab), Objectifs (addGoal/setCurrent). Cause : le singleton dialogs exportait confirm et alert mais avait oublié prompt, alors que Dialog.prompt() existait déjà dans ui-premium/Dialog.js depuis longtemps.

// 0.58.44 - app/dialogs.js
export const dialogs = {
  confirm(options) { /* ... */ },
  alert(options) { /* ... */ },
  // ✅ AJOUTÉ
  prompt(options) {
    const opts = options || {};
    const newAttempt = tryNewDialog('prompt', {
      title: opts.title || 'Saisie',
      message: opts.message,
      defaultValue: opts.defaultValue || '',
      placeholder: opts.placeholder || '',
      okLabel: opts.okLabel,
      cancelLabel: opts.cancelLabel,
    });
    if (newAttempt) return newAttempt;
    // Fallback : window.prompt natif si Dialog premium indispo
    if (typeof window !== 'undefined' && window.prompt) {
      const msg = [opts.title, opts.message].filter(Boolean).join('\n');
      const v = window.prompt(msg || 'Valeur :', opts.defaultValue || '');
      return Promise.resolve(v);
    }
    return Promise.resolve(null);
  },
};

🎁 3. Bannière widgets bonus + badge "Personnaliser"

Pourquoi : tu ne voyais pas les widgets car ils sont opt-in (désactivés par défaut). Maintenant ils sont visibles et accessibles via une bannière d'invitation + un badge sur le bouton.

🎁 6 widgets bonus disponibles : " Citation, 📅 Mini calendrier, 🔗 Liens favoris, ☁ Météo, 📝 Mes notes, 🎯 Mes objectifs 
// 0.58.44 - app/accueil/page.js

// 1) Badge sur bouton "Personnaliser"
const hiddenOptIn = ALL_WIDGETS.filter(w => !widgets[w.id]);

<button onClick={() => setEditLayout(!editLayout)} style={{ position: 'relative' }}>
  <i className='ti ti-layout-dashboard' /> Personnaliser
  {!editLayout && hiddenOptIn.length > 0 && (
    <span style={{
      position: 'absolute', top: -6, right: -6,
      background: 'linear-gradient(135deg, #EF9F27, #d28818)',
      color: '#fff', minWidth: 18, height: 18, borderRadius: 9,
      border: '2px solid #fff',
    }}>{hiddenOptIn.length}</span>
  )}
</button>

// 2) Bannière d'invitation (dismissable)
const [bannerDismissed, setBannerDismissed] = useState(true);
useEffect(() => {
  setBannerDismissed(localStorage.getItem('av-widgets-banner-dismissed') === 'true');
}, []);

{!editLayout && !bannerDismissed && hiddenOptIn.length > 0 && (
  <div>
    🎁 <b>{hiddenOptIn.length} widgets bonus disponibles</b> :
    {hiddenOptIn.slice(0, 6).map(w => <span><i className={w.icon} />{w.label}</span>)}
    <button onClick={() => setEditLayout(true)}>✨ Découvrir</button>
    <button onClick={() => { localStorage.setItem('av-widgets-banner-dismissed', 'true'); }}>✕</button>
  </div>
)}
