Bouton crayon bleu au hover sur chaque lien favori, à côté de la croix rouge. Pré-remplit le libellé et l'URL pour modification rapide.

// 0.58.39 - app/components/DashboardWidgets.js
async function editFav(idx) {
  const current = favs[idx];
  const label = await dialogs.prompt({
    title: 'Modifier le libellé',
    defaultValue: current.label,  // ← pré-rempli
  });
  if (label === null) return;
  const url = await dialogs.prompt({
    title: "Modifier l'URL",
    defaultValue: current.url,
  });
  if (url === null) return;
  const next = favs.map((f, i) => i === idx ? { label, url } : f);
  setFavs(next);
  setFavLinks(next);
}

🪝 4. Hook useCurrentContext + Filtre /interventions

Hook réutilisable qui partage le contexte bât/svc défini dans la TopBar. Appliqué à /interventions avec chargement des patient_ids du contexte via chambres → patients.

// 0.58.39 - lib/useCurrentContext.js (NEW)
export function useCurrentContext() {
  const [state, setState] = useState({ batimentId: null, serviceId: null, active: false });

  useEffect(() => {
    // Lit localStorage + écoute event av-current-context-change
    const batimentId = localStorage.getItem('av-current-batiment-id');
    const serviceId = localStorage.getItem('av-current-service-id');
    setState(prev => ({ ...prev, batimentId, serviceId }));

    function onCtxChange(e) { setState(prev => ({ ...prev, ...e.detail })); }
    window.addEventListener('av-current-context-change', onCtxChange);
    return () => window.removeEventListener('av-current-context-change', onCtxChange);
  }, []);

  return { ...state, toggle: () => setState(p => ({...p, active: !p.active})), setActive };
}

// app/interventions/page.js
const ctx = useCurrentContext();
const [ctxPatientIds, setCtxPatientIds] = useState(null);

useEffect(() => {
  if (!ctx.batimentId && !ctx.serviceId) { setCtxPatientIds(null); return; }
  (async () => {
    // chambres du contexte → patient_ids
    let query = supabase.from('chambres').select('id, service_id, batiment_id');
    if (ctx.serviceId) query = query.eq('service_id', ctx.serviceId);
    else if (ctx.batimentId) query = query.eq('batiment_id', ctx.batimentId);
    const { data: chambres } = await query;
    const { data: pats } = await supabase.from('patients').select('id')
                                  .in('chambre_id', chambres.map(c => c.id));
    setCtxPatientIds(new Set(pats.map(p => p.id)));
  })();
}, [ctx.batimentId, ctx.serviceId]);

// Filtrage
const visible = rows.filter(r =>
  (!fStatut || r.statut === fStatut) &&
  (!fType || r.type === fType) &&
  (!ctx.active || !ctxPatientIds || ctxPatientIds.has(r.patient_id))
);

function formatRelative(ts) {
  const diff = Date.now() - ts;
  const sec = Math.floor(diff / 1000);
  if (sec < 60) return "à l'instant";
  const min = Math.floor(sec / 60);
  if (min < 60) return `il y a ${min}min`;
  const h = Math.floor(min / 60);
  if (h < 24) return `il y a ${h}h`;
  const d = Math.floor(h / 24);
  if (d === 1) return 'hier';
  return `il y a ${d}j`;
}
