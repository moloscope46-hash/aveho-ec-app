Tu avais reçu la version corrigée séparément qui a été exécutée en base avec succès, mais le repo gardait l'ancien fichier. C'est maintenant aligné avec ce qui est réellement en base.

-- 0.58.29 : repo synchronisé avec ce qui est en base
CREATE POLICY "audit_log_insert_v2" ON audit_log
FOR INSERT
TO authenticated
WITH CHECK (
  user_id = auth.uid()
  AND
  EXISTS (
    SELECT 1 FROM membres_structure  -- ✓ singulier (vrai nom de table)
    WHERE membres_structure.user_id = auth.uid()
      AND membres_structure.structure_id = audit_log.structure_id
      AND (membres_structure.actif IS NULL OR membres_structure.actif = true)
      --                ^^^^^ ✓ colonne 'actif' (booléen), pas 'archive'
  )
);
