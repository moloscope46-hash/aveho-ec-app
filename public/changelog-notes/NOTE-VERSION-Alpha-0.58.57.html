// 0.58.57 - app/components/DashboardWidgets.js
export function TeamGoalsWidget() {
  const [teamGoals, setTeamGoals] = useState([]);
  
  useEffect(() => {
    (async () => {
      // 1) Mes équipes
      const { data: memb } = await supabase.from('membres_equipe')
        .select('equipe_id, equipes(id, nom, couleur)')
        .eq('user_id', user.id);
      const teamIds = memb.map(m => m.equipe_id);
      
      // 2) Objectifs partagés (RLS protège : on ne voit que les goals partagés avec nos équipes)
      const { data: goals } = await supabase.from('user_goals')
        .select('*')
        .eq('shared', true)
        .in('team_id', teamIds)
        .neq('user_id', user.id);  // exclut les miens
      
      // 3) Enrich avec nom du propriétaire
      const ownerIds = [...new Set(goals.map(g => g.user_id))];
      const { data: mss } = await supabase.from('membres_structure')
        .select('user_id, nom_affiche')
        .in('user_id', ownerIds);
      
      setTeamGoals(goals.map(g => ({ ...g, ownerName: ... })));
    })();
  }, []);
}

🚚 4. Filtre ctx étendu à /transferts

La page /transferts utilise maintenant useCurrentContext() et charge les depot_ids du bâtiment actif via la nouvelle colonne depots.batiment_id. Filtrage : conserve les transferts depot↔depot dans le bâtiment + tous les transferts chambre↔chambre. 8 pages filtrent maintenant par ctx (interventions, materiels, maintenance, patients, commandes, achats, signalements, transferts).

⚙ Filtré sur le bâtiment (3 dépôts) ← indicateur visuel quand le filtre est actif

// 0.58.57 - app/transferts/page.js
const ctx = useCurrentContext();
const [depotsCtx, setDepotsCtx] = useState(null);

useEffect(() => {
  if (!ctx.active || !ctx.batimentId) {
    setDepotsCtx(null);
    return;
  }
  // Charge les dépôts du bâtiment via la nouvelle colonne batiment_id
  supabase.from('depots').select('id').eq('batiment_id', ctx.batimentId)
    .then(({ data }) => setDepotsCtx(new Set(data.map(d => d.id))));
}, [ctx.active, ctx.batimentId]);

const rowsFiltered = ctx.active && depotsCtx
  ? rows.filter(r => {
      const srcMatch = r.src_type === 'depot' && depotsCtx.has(r.src_id);
      const dstMatch = r.dst_type === 'depot' && depotsCtx.has(r.dst_id);
      // Chambre↔chambre : garde tout
      if (r.src_type !== 'depot' && r.dst_type !== 'depot') return true;
      return srcMatch || dstMatch;
    })
  : rows;
