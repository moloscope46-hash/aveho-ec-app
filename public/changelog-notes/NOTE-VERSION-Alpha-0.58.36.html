👤 2. UserMenu : bâtiments + équipes rattachés sous le nom

Sous le nom, l'email et le rôle, on voit maintenant les bâtiments et équipes auxquels l'utilisateur est rattaché. Ces infos sont chargées via membres_equipe → equipes → batiments, filtrées sur l'établissement courant. Affichage silencieux si aucune équipe.

// 0.58.36 - app/components/UserAttachmentsInfo.js (NEW)
export default function UserAttachmentsInfo({ userId, etabId }) {
  const [batiments, setBatiments] = useState([]);
  const [equipes, setEquipes] = useState([]);

  useEffect(() => {
    const { data: memb } = await supabase
      .from('membres_equipe')
      .select('equipe_id, equipes(id, nom, couleur, batiment_id, batiments(id, nom, etablissement_id))')
      .eq('user_id', userId);

    // Filtre par établissement courant (+ transversales sans bâtiment)
    const eqs = memb.map(m => m.equipes).filter(eq =>
      eq.batiments?.etablissement_id === etabId || !eq.batiment_id
    );

    // Bâtiments uniques agrégés depuis les équipes
    const batMap = {};
    eqs.forEach(eq => eq.batiments && (batMap[eq.batiments.id] = eq.batiments));
    setBatiments(Object.values(batMap));
    setEquipes(eqs);
  }, [userId, etabId]);

  if (equipes.length === 0 && batiments.length === 0) return null;  // silencieux

  return (
    <div>
      {batiments.length > 0 && (
        <div><i className='ti ti-building' /> {batiments.map(b => b.nom).join(' · ')}</div>
      )}
      {equipes.length > 0 && (
        <div><i className='ti ti-users' /> {equipes.map(eq => eq.nom).slice(0, 3).join(' · ')}
          {equipes.length > 3 && <span> +{equipes.length - 3}</span>}
        </div>
      )}
    </div>
  );
}

🩹 3. Fix 400 consentements_rgpd défensif

Console : ?select=id,date_signature,a_consenti,date_expiration → 400. Une colonne manque en base. Query passée en défensive avec try/catch + fallback à id, date_signature minimal. Comme on a fait avec template_libelle en 0.56.17.

// 0.58.36 - app/patient/[id]/page.js
// Query défensive (IIFE dans Promise.all)
(async () => {
  try {
    const r = await supabase.from('consentements_rgpd')
      .select('id, date_signature, a_consenti, date_expiration')
      .eq('patient_id', patId)
      .order('date_signature', { ascending: false });
    if (r.error) throw r.error;
    return r;
  } catch {
    // Fallback : colonnes manquantes en base
    try {
      const r = await supabase.from('consentements_rgpd')
        .select('id, date_signature')
        .eq('patient_id', patId)
        .order('date_signature', { ascending: false });
      return r;
    } catch { return { data: [] }; }
  }
})(),
