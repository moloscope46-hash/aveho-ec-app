🎯 2. Le composant IconPicker (sélecteur réutilisable)

Nouveau composant IconPicker avec champ de recherche live, catégories pliables, bouton "Aucune icône" pour revenir au défaut. Réutilisable partout dans l'app (tags, étiquettes, annonces, et facilement extensible).

Médical & soins
Matériel & équipement
// 0.58.47 - app/components/IconPicker.js (NEW)
export const ICON_CATALOG = {
  'Tags & marqueurs': ['ti-tag', 'ti-bookmark', 'ti-star', 'ti-flag', 'ti-pin', 'ti-heart'],
  'Statut & état':    ['ti-check', 'ti-x', 'ti-alert-triangle', 'ti-shield-check', 'ti-shield-x'],
  'Médical & soins':  ['ti-stethoscope', 'ti-heartbeat', 'ti-medical-cross', 'ti-pill', 'ti-vaccine', 'ti-bandage', 'ti-mask', 'ti-emergency-bed', 'ti-wheelchair'],
  'Matériel & équipement': ['ti-armchair-2', 'ti-bed', 'ti-tools', 'ti-screwdriver', 'ti-settings', 'ti-package', 'ti-box', 'ti-barcode', 'ti-device-tablet'],
  'Urgence & priorité': ['ti-bolt', 'ti-flame', 'ti-clock-hour-3', 'ti-hourglass-high', 'ti-urgent', 'ti-fire-extinguisher'],
  'Communication':    ['ti-bell', 'ti-mail', 'ti-phone', 'ti-message-circle', 'ti-megaphone'],
  'Personnes':        ['ti-user', 'ti-users', 'ti-user-plus', 'ti-user-check', 'ti-user-star'],
  'Lieux':            ['ti-home', 'ti-building-hospital', 'ti-building-warehouse', 'ti-map-pin'],
  'Documents':        ['ti-file-text', 'ti-file-check', 'ti-clipboard', 'ti-receipt', 'ti-certificate'],
  'Symboles':         ['ti-circle-filled', 'ti-rosette', 'ti-sparkles', 'ti-crown', 'ti-rocket', 'ti-target'],
};

export default function IconPicker({ value, onChange, color = '#185FA5', allowEmpty = true }) {
  const [query, setQuery] = useState('');
  const filteredCatalog = !query ? ICON_CATALOG : (...);

  return (
    <div>
      <input placeholder='Rechercher…' value={query} onChange={...} />
      {allowEmpty && <button onClick={() => onChange(null)}>Aucune icône</button>}

      {Object.entries(filteredCatalog).map(([cat, icons]) => (
        <div>
          <h4>{cat}</h4>
          {icons.map(icon => (
            <button onClick={() => onChange(icon)}
              style={{ border: value === icon ? `2px solid ${color}` : '1px solid #e3e9ee' }}>
              <i className={`ti ${icon}`} />
            </button>
          ))}
        </div>
      ))}
    </div>
  );
}

📦 5. Migration SQL à exécuter

À copier-coller dans Supabase SQL Editor (ou utilise le fichier public/sql/migration-0.58.47-tags-icone.sql) :

ALTER TABLE tags_materiel
ADD COLUMN IF NOT EXISTS icone TEXT;

ALTER TABLE etiquettes
ADD COLUMN IF NOT EXISTS icone TEXT;

ALTER TABLE annonces
ADD COLUMN IF NOT EXISTS icone TEXT;

⚠ Sans cette migration, l'app continue de fonctionner mais le champ icone sera ignoré par Supabase — tous les tags garderont l'icône par défaut.
