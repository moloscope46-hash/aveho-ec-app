const [bulkProgress, setBulkProgress] = useState(null);

async function bulkExport() {
  const total = selected.size;
  let current = 0;
  for (const id of selected) {
    setBulkProgress({ current, total, label: 'Export en cours...' });
    await exportItem(id);
    current++;
  }
  setBulkProgress(null);
}

🎥 Mode présentation pour démos clients

Raccourci Ctrl+Shift+P (ou Cmd+Shift+P sur Mac) — toggle le mode présentation. Effets :

  • Zoom léger 1.05x (font-size 17.5px = meilleure lisibilité projection)
  • Animations ralenties 1.5x sur tout l'app (effet "showroom" hypnotique)
  • Ombres et glow renforcés sur panels/KPIs
  • Cache les badges debug et version
  • Affiche un badge "MODE PRÉSENTATION" en bas (pulse)
  • Persisté dans localStorage (reload garde le mode)
