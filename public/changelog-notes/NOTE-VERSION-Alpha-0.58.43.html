📊 1. Exports CSV factorisés sur 3 pages

Cmd+K → tape export csv → ça déclenche l'export sur la page courante. 4 pages branchées : /patients (depuis 0.58.42), /interventions, /materiels, /maintenance.

Les exports respectent automatiquement les filtres actifs (statut, type, contexte bât/svc). Tu actives le filtre contexte → l'export ne contient que les lignes du bâtiment courant.

// 0.58.43 - /interventions/page.js
async function exportInterventionsCsv() {
  // Respecte les mêmes filtres que la vue
  const data = rows.filter(r => {
    if (fStatut && r.statut !== fStatut) return false;
    if (fType && r.type !== fType) return false;
    if (ctx.active && ctxPatientIds && !ctxPatientIds.has(r.patient_id)) return false;
    return true;
  });
  const { exportRows } = await import('../../lib/exportExcel');
  await exportRows(data, {
    filename: `interventions_${date}`,
    columns: { Numéro, Date, Type, Urgence, Statut, Matériel, Patient, Chambre, Description }
  });
}
usePageAction('export-csv', () => exportInterventionsCsv());

⌨️ 2. Notes : raccourci Ctrl+Shift+X

En mode édition, presser Ctrl+Shift+X (ou ⌘+Shift+X sur Mac) sur une ligne effectue 3 actions intelligentes selon le contenu :

📝 Mode édition

## Sprint 0.58 - truc à faire| - [ ] visites clients - [x] export CSV factorisé
💡 Curseur sur "truc à faire" → Ctrl+Shift+X = devient "- [ ] truc à faire" 62 / 4000

Comportement détecté automatiquement :

  • - [ ] ou - [x] → toggle l'état
  • - bullet normale → convertit en - [ ]
  • Ligne de texte → convertit en - [ ] texte
  • Ligne vide → insère - [ ]

const GOALS_STORAGE_KEY = 'av-personal-goals';

const GOAL_COLORS = [
  { id: 'navy',  bg: '#142131', grad: 'linear-gradient(90deg, #142131, #2a3850)' },
  { id: 'teal',  bg: '#7CC8C8', grad: '...' },
  { id: 'terra', bg: '#C9867F', grad: '...' },
  { id: 'amber', bg: '#EF9F27', grad: '...' },
  { id: 'green', bg: '#5aa05a', grad: '...' },
  { id: 'blue',  bg: '#185FA5', grad: '...' },
];

export function ObjectifsWidget() {
  const [goals, setGoals] = useState([]);

  return goals.map(g => {
    const pct = Math.min(100, (g.current / g.target) * 100);
    const isComplete = g.current >= g.target;
    return (
      <div>
        <div className='progress'>
          <div style={{
            width: `${pct}%`,
            background: isComplete ? 'linear-gradient(90deg, #5aa05a, #4a8a4a)' : color.grad,
            boxShadow: isComplete ? '0 0 8px rgba(90,160,90,.50)' : `0 0 6px ${color.bg}40`,
          }} />
          {/* Milestones 25/50/75 */}
          {[25, 50, 75].map(m => <div style={{ left: `${m}%` }} />)}
        </div>
        {/* Boutons rapides */}
        <button onClick={() => quickIncrement(g.id, 1)}>+1</button>
        <button onClick={() => quickIncrement(g.id, 5)}>+5</button>
      </div>
    );
  });
}
