🩹 1. Fix 404 prod — vues stats DI

Bug observé :

GET https://[supabase]/rest/v1/v_stats_di_top_demandeurs?...&order=nb_di.desc&limit=10 404 (Not Found)

La vue est définie dans aveho-PATCH-vers-0.43.sql mais n'a apparemment jamais été déployée sur l'instance Supabase de prod. Fix en 2 étapes :

A. SQL : (re)création des 6 vues stats DI

À copier-coller dans Supabase SQL Editor (ou utilise le fichier public/sql/migration-0.58.48-stats-di-views.sql) :

CREATE OR REPLACE VIEW v_stats_di_global AS ...
CREATE OR REPLACE VIEW v_stats_di_par_type AS ...

CREATE OR REPLACE VIEW v_stats_di_heatmap AS ...
CREATE OR REPLACE VIEW v_stats_di_top_demandeurs AS ...  ← celle qui manquait
CREATE OR REPLACE VIEW v_stats_di_par_mois AS ...

CREATE OR REPLACE → safe sur les vues existantes (juste un refresh de la définition), crée celles manquantes.

B. Front : helper safeViewQuery pour swallow les 404

Si à l'avenir une autre vue manque, plus de GET 404 dans la console — juste un console.warn discret et les autres stats continuent de s'afficher.

[stats] Vue 'v_stats_di_top_demandeurs' indisponible (migration manquante). Retour [].
// 0.58.48 - app/statistiques-interventions/page.js
async function safeViewQuery(viewName, builderFn) {
  try {
    const builder = supabase.from(viewName).select('*').eq('structure_id', auth.structureId);
    const finalQuery = builderFn ? builderFn(builder) : builder;
    const { data, error } = await finalQuery;
    if (error) {
      // 404 vue manquante → warn discret, pas de GET 404 dans la console
      if (error.code === 'PGRST205' || error.code === '42P01' || /not found/i.test(error.message || '')) {
        console.warn(`[stats] Vue '${viewName}' indisponible (migration manquante). Retour [].`);
        return null;
      }
      throw error;
    }
    return data;
  } catch (e) {
    console.warn(`[stats] Erreur vue '${viewName}':`, e?.message || e);
    return null;
  }
}

// Tous les appels passent par le helper
const [gResp, tResp, uResp, hResp, dResp, mResp] = await Promise.all([
  safeViewQuery('v_stats_di_global', b => b.maybeSingle()),
  safeViewQuery('v_stats_di_par_type', b => b.order('nb_total', { ascending: false }).limit(10)),
  safeViewQuery('v_stats_di_par_urgence'),
  safeViewQuery('v_stats_di_heatmap'),
  safeViewQuery('v_stats_di_top_demandeurs', b => b.order('nb_di', { ascending: false }).limit(10)),
  safeViewQuery('v_stats_di_par_mois', b => b.order('mois_debut')),
]);

const SUGGESTION_KEYWORDS = {
  // Médical
  'ti-stethoscope': ['stetho', 'consult', 'examen', 'diagnos', 'auscult'],
  'ti-heartbeat':   ['cardiaque', 'coeur', 'cardio', 'ecg'],
  'ti-pill':        ['medicament', 'traitement', 'comprime', 'pill'],
  'ti-vaccine':     ['vaccin', 'injection', 'piqure', 'seringue'],
  'ti-bandage':     ['pansement', 'cicatris', 'plaie', 'blessure'],
  'ti-emergency-bed': ['hospital', 'hospitalis', 'urgence', 'samu'],
  'ti-wheelchair':  ['fauteuil', 'mobilité', 'handicap', 'vph', 'roulant'],
  // Matériel
  'ti-tools':       ['maintenance', 'outil', 'réparation', 'sav'],
  'ti-package':     ['livraison', 'colis', 'expédition'],
  'ti-barcode':     ['code-barre', 'scan', 'gs1', 'traçabilité'],
  // Urgence
  'ti-bolt':        ['urgent', 'critique', 'express', 'priorité'],
  'ti-flame':       ['incendie', 'feu', 'brulure'],
  // Statut
  'ti-shield-check':['sous garantie', 'garantie', 'protégé', 'sécurisé'],
  // ... ~40 icônes au total
};

export function suggestIcon(label) {
  if (!label) return null;
  const normalized = label.toLowerCase().trim();
  // Trie par longueur de keyword (longest match first)
  // 'sous garantie' (13 chars) matche AVANT 'garantie' (8 chars)
  const entries = Object.entries(SUGGESTION_KEYWORDS).flatMap(([icon, keywords]) =>
    keywords.map(kw => ({ icon, kw, len: kw.length }))
  );
  entries.sort((a, b) => b.len - a.len);
  for (const { icon, kw } of entries) {
    if (normalized.includes(kw)) return icon;
  }
  return null;
}
