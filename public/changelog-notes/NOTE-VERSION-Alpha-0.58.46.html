🎛 1. Drag&drop colonnes — tous les tableaux Crud

Sur tous les tableaux Crud du projet (matériels, articles, dépôts, types intervention, etc.), tu peux maintenant réordonner les colonnes à la souris. L'ordre est sauvegardé par table en localStorage. Une nouvelle colonne ajoutée plus tard apparaîtra à la fin de ton ordre custom — pas besoin de tout refaire.

const colsStorageKey = `av-crud-cols-${table}`;  // par table
const [colOrder, setColOrder] = useState(null);
const [draggedColKey, setDraggedColKey] = useState(null);
const [dragOverColKey, setDragOverColKey] = useState(null);

// Réconciliation intelligente au mount
useEffect(() => {
  const stored = JSON.parse(localStorage.getItem(colsStorageKey) || '[]');
  const currentKeys = columns.map(c => c.key);
  // Garde l'ordre stocké pour les colonnes qui existent encore
  const validStored = stored.filter(k => currentKeys.includes(k));
  // Ajoute les nouvelles colonnes (ajoutées en dev plus tard) à la fin
  const missing = currentKeys.filter(k => !validStored.includes(k));
  setColOrder([...validStored, ...missing]);
}, [colsStorageKey, columns.length]);

const orderedColumns = colOrder
  ? colOrder.map(k => columns.find(c => c.key === k)).filter(Boolean)
  : columns;

// Handler drop : swap
function handleColDrop(e, targetKey) {
  e.preventDefault();
  setColOrder(prev => {
    const arr = [...prev];
    const fromIdx = arr.indexOf(draggedColKey);
    const toIdx = arr.indexOf(targetKey);
    arr.splice(fromIdx, 1);
    arr.splice(toIdx, 0, draggedColKey);
    return arr;
  });
}

// 0.58.46 - clic + swipe touch
const [focusForecastIdx, setFocusForecastIdx] = useState(null);
const touchStartX = useRef(null);

<div
  onTouchStart={(e) => { touchStartX.current = e.touches[0].clientX; }}
  onTouchEnd={(e) => {
    const dx = e.changedTouches[0].clientX - touchStartX.current;
    if (Math.abs(dx) < 40) return;  // pas un swipe
    const dir = dx < 0 ? 1 : -1;     // gauche = jour suivant
    setFocusForecastIdx(curr => {
      const next = (curr ?? 0) + dir;
      if (next < 0 || next >= 3) return curr;
      return next;
    });
  }}
>
  {state.forecast.time.slice(1, 4).map((iso, i) => (
    <div onClick={() => setFocusForecastIdx(focusForecastIdx === i ? null : i)}
         style={{
           transform: focusForecastIdx === i ? 'scale(1.04)' : 'scale(1)',
           border: focusForecastIdx === i ? '1px solid #185FA5' : '1px solid rgba(188,224,247,.5)',
         }}>
      ...
    </div>
  ))}
</div>

{focusForecastIdx !== null && (
  <div className='detail-panel'>
    Vent max · Précip · UV · Lever · Coucher
  </div>
)}
