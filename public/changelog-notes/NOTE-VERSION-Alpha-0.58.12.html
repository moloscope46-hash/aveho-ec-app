API impérative qui retourne une Promise — au-dessus de Modal mais ergonomique. Plus besoin de boilerplate useState pour les confirmations simples.

Dialog.alert{title, message, variant: info/success/warning/danger}Promise<void>
// AVANT - boilerplate verbeux
const [showConfirm, setShowConfirm] = useState(false);
const [pendingPatient, setPendingPatient] = useState(null);
function askDelete(p) { setPendingPatient(p); setShowConfirm(true); }
<Modal open={showConfirm} ...>...</Modal>

// APRÈS - impératif
const ok = await Dialog.confirm({
  title: 'Supprimer ce patient ?',
  message: 'Cette action est irréversible.',
  danger: true,
  preview: `${p.nom} ${p.prenom} — Chambre ${p.chambre}`,
});
if (ok) {
  await deletePatient(p.id);
  toast.undo('Patient supprimé', () => restore(p));
}

// Prompt avec validation
const reason = await Dialog.prompt({
  title: 'Motif de rejet',
  placeholder: 'Expliquer pourquoi…',
  validate: (v) => v.length < 10 ? 'Au moins 10 caractères' : null,
  multiline: true,    // textarea + Cmd/Ctrl+Enter pour valider
});
