🚨 1. FIX build Vercel (nested-styled-jsx)

Symptôme : Vercel échouait avec Two ChildStyledJsx exist in one ChildStyledJsx sur FloatingActionBar.js:312.

Cause : Next.js 15 + SWC interdit deux <style jsx global> dans le même composant. On en avait un au top-level + un dans le bloc {popupUrl && (...)}.

// AVANT 0.58.62 - 2 blocs styled-jsx (cassé sur Next.js 15)
return (
  <>
    <div className="av-shortcuts-bar">...</div>

    <style jsx global>{`        // ← Bloc #1 (top-level)
      @keyframes av-fab-pulse {...}
    `}</style>

    {popupUrl && (
      <div>
        <iframe ... />
        <style jsx global>{`      // ← Bloc #2 (NESTED, interdit)
          @keyframes av-popup-fade-in {...}
        `}</style>
      </div>
    )}
  </>
);

// APRÈS 0.58.62 - un seul bloc consolidé
return (
  <>
    <div>...</div>
    {popupUrl && <div><iframe ... /></div>}

    <style jsx global>{`        // ← bloc unique avec tous les keyframes
      @keyframes av-fab-pulse {...}
      @keyframes av-popup-fade-in {...}     // ← fusionné ici
      @media (max-width: 768px) {...}
    `}</style>
  </>
);

🪝 4. Hook useContextPatientIds étendu avec 3 modes

// lib/useContextPatientIds.js (0.58.62)
export function useContextPatientIds() {
  const ctx = useCurrentContext();
  // ...
  useEffect(() => {
    if (!ctx.active || (!ctx.batimentId && !ctx.serviceId && !ctx.equipeId)) {
      setPatientIds(null);
      return;
    }

    // Mode 1 : équipe seule → direct par equipe_id
    if (ctx.equipeId && !ctx.serviceId && !ctx.batimentId) {
      const { data: pats } = await supabase
        .from('patients').select('id').eq('equipe_id', ctx.equipeId);
      setPatientIds(new Set(pats.map(p => p.id)));
      return;
    }

    // Mode 2 + 3 : chambres puis patients (avec ou sans intersect équipe)
    const chambres = await loadChambresInPerimeter();
    let patQuery = supabase.from('patients').select('id, equipe_id')
      .in('chambre_id', chambres.map(c => c.id));
    if (ctx.equipeId) patQuery = patQuery.eq('equipe_id', ctx.equipeId);
    const { data: pats } = await patQuery;
    setPatientIds(new Set(pats.map(p => p.id)));
  }, [ctx.active, ctx.batimentId, ctx.serviceId, ctx.equipeId]);
}
