📋 2. Notes : Ctrl+V pour insérer une image

En mode édition de note, colle directement une capture d'écran (Win+Shift+S, ⌘+Shift+4) ou n'importe quelle image du presse-papier. Elle est compressée auto via canvas (max 800px, qualité 0.78) puis insérée en markdown ![alt](data:image/jpeg;base64,...) à la position du curseur. Limite par image : 250 Ko (auto re-compression à qualité réduite si dépasse).

// 0.58.45 - Compression auto via canvas
async function compressImageToDataUrl(blob) {
  return new Promise(resolve => {
    const reader = new FileReader();
    reader.onload = () => {
      const img = new Image();
      img.onload = () => {
        const ratio = img.width > 800 ? 800 / img.width : 1;
        const canvas = document.createElement('canvas');
        canvas.width = img.width * ratio;
        canvas.height = img.height * ratio;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = '#fff';  // fond blanc pour les PNG transparents
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
        let dataUrl = canvas.toDataURL('image/jpeg', 0.78);
        // Re-tente avec qualité plus basse si dépasse 250 Ko
        if (estimateKB(dataUrl) > 250) {
          dataUrl = canvas.toDataURL('image/jpeg', 0.55);
        }
        resolve(estimateKB(dataUrl) > 250 ? null : dataUrl);
      };
      img.src = reader.result;
    };
    reader.readAsDataURL(blob);
  });
}

// Textarea onPaste
<textarea
  onPaste={async (e) => {
    const imageItem = Array.from(e.clipboardData.items)
      .find(it => it.type.startsWith('image/'));
    if (!imageItem) return;  // collage texte normal
    e.preventDefault();
    const dataUrl = await compressImageToDataUrl(imageItem.getAsFile());
    if (!dataUrl) { dialogs.alert({ title: 'Image trop volumineuse' }); return; }
    const pos = e.target.selectionStart;
    const md = `![Image collée](${dataUrl})`;
    updateActiveText(text.slice(0, pos) + md + text.slice(pos));
  }}
/>
