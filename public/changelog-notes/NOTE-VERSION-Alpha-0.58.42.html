☑️ 2. Checkboxes to-do interactives

Syntaxe markdown : - [ ] truc à faire (vide) ou - [x] truc fait (coché). En mode preview, cliquer sur une case toggle l'état directement dans le texte source — pas besoin de passer en édition.

// 0.58.42 - mini-parser étendu pour checkboxes
const allCheckboxes = lines.every(l =>
  /^-\s*\[[\sxX]\]\s/.test(l.trim())
);

if (allCheckboxes) {
  return (
    <ul style={{ listStyle: 'none' }}>
      {lines.map(l => {
        const m = l.trim().match(/^-\s*\[([\sxX])\]\s(.*)$/);
        const checked = m[1].toLowerCase() === 'x';
        return (
          <li>
            <span onClick={() => onToggleCheckbox(lineIdx)}
                  style={{
                    border: `2px solid ${checked ? '#5aa05a' : '#d4c896'}`,
                    background: checked ? '#5aa05a' : 'transparent',
                    cursor: 'pointer',
                  }}>
              {checked && '✓'}
            </span>
            <span style={{
              textDecoration: checked ? 'line-through' : 'none',
              color: checked ? '#8a98a8' : '#142131',
            }}>{content}</span>
          </li>
        );
      })}
    </ul>
  );
}

// Fonction toggleCheckbox : flip ' ' <-> 'x' à l'index N
function toggleCheckbox(targetIdx) {
  let cbIdx = -1;
  const updated = text.split('\n').map(line => {
    const m = line.match(/^(\s*-\s*\[)([\sxX])(\]\s.*)$/);
    if (m) {
      cbIdx++;
      if (cbIdx === targetIdx) {
        return m[1] + (m[2].toLowerCase() === 'x' ? ' ' : 'x') + m[3];
      }
    }
    return line;
  });
  updateActiveText(updated.join('\n'));
}

🔍 3. Filtre contexte sur /maintenance (4e page)

Pattern hooks étendu à /maintenance. Particularité : 3 niveaux de jointure Supabase car les maintenances sont rattachées à des matériels qui sont rattachés à des patients qui sont dans des chambres :

chambres → patients → matériels → maintenances

// 0.58.42 - app/maintenance/page.js
const ctx = useCurrentContext();
const [ctxMaterielIds, setCtxMaterielIds] = useState(null);

useEffect(() => {
  if (!ctx.batimentId && !ctx.serviceId) { setCtxMaterielIds(null); return; }
  (async () => {
    // 1) Chambres du contexte
    const { data: chambres } = await supabase.from('chambres').select('id')
      .eq(ctx.serviceId ? 'service_id' : 'batiment_id', ctx.serviceId || ctx.batimentId);

    // 2) Patients dans ces chambres
    const { data: pats } = await supabase.from('patients').select('id')
      .in('chambre_id', chambres.map(c => c.id));

    // 3) Matériels assignés à ces patients
    const { data: mats } = await supabase.from('materiels').select('id')
      .in('patient_id', pats.map(p => p.id));

    setCtxMaterielIds(new Set(mats.map(m => m.id)));
  })();
}, [ctx.batimentId, ctx.serviceId]);

// Filtre
const filtered = (() => {
  let r = rows;
  if (fStatut) r = r.filter(row => statutEffectif(row) === fStatut);
  if (ctx.active && ctxMaterielIds) {
    r = r.filter(row => row.materiel_id && ctxMaterielIds.has(row.materiel_id));
  }
  return r;
})();

🪝 5. Hook usePageAction (event bus)

Pattern découplé via window.dispatchEvent. Cmd+K émet, les pages écoutent. Pas besoin de modifier GlobalSearch quand on ajoute une action.

// 0.58.42 - lib/usePageAction.js (NEW)
import { useEffect } from "react";

export function usePageAction(actionName, handler) {
  useEffect(() => {
    if (!actionName || !handler) return;
    function onPageAction(e) {
      if (e?.detail?.action === actionName) {
        handler();
      }
    }
    window.addEventListener("av-page-action", onPageAction);
    return () => window.removeEventListener("av-page-action", onPageAction);
  }, [actionName, handler]);
}

// Côté GlobalSearch : émetteur
if (a.url.startsWith("#page-action:")) {
  const actionName = a.url.slice("#page-action:".length);
  window.dispatchEvent(new CustomEvent("av-page-action", {
    detail: { action: actionName, path: pathname },
  }));
  setOpen(false);
  return;
}

// Côté pages : écouteur
usePageAction("open-new", () => openNew());
usePageAction("export-csv", () => exportPatientsCsv());
usePageAction("toggle-ctx-filter", () => ctx.toggle());
