const ACTIONS = [
  { id: 'new-patient', lbl: 'Créer un patient', ...,
    pageContext: /^\/patients/ },
  { id: 'new-achat', lbl: 'Nouvelle demande d\'achat', ...,
    pageContext: /^\/(achats|commandes)/ },
  { id: 'toggle-focus', url: '#toggle-focus',
    keywords: ['focus', 'zen', 'concentration'] },
  { id: 'toggle-presentation', url: '#toggle-presentation',
    keywords: ['présentation', 'démo'] },
  { id: 'goto-historique', url: '/historique',
    keywords: ['historique', 'audit', 'log'] },
];

// findActions trie par pertinence contextuelle
function findActions(query, currentPath) {
  // Pas de query + page connue → actions contextuelles direct
  if (!query && currentPath) {
    return ACTIONS.filter(a =>
      a.pageContext && a.pageContext.test(currentPath)
    ).slice(0, 4);
  }
  // Sinon, filtre + tri par contexte
  const matches = ACTIONS.filter(a => match(a, query));
  matches.sort((a, b) => {
    const aMatch = a.pageContext?.test(currentPath) ? 1 : 0;
    const bMatch = b.pageContext?.test(currentPath) ? 1 : 0;
    return bMatch - aMatch;
  });
  return matches.slice(0, 6);
}

🧘 Mode focus zen pour saisie concentrée

Raccourci Ctrl+Shift+F — toggle le mode focus. Effets :

  • Cache topbar, sidebar, notif badges, alert banners, toasts
  • Centre le contenu (max-width 800px)
  • Atténue breadcrumbs/version-badge (opacity 0.3 → 1 au hover)
  • Badge discret "FOCUS ZEN" en haut à droite
  • Persisté localStorage (reload garde le mode)

// 0.58.25 - lib/focusMode.js
export function toggleFocusMode() {
  const next = !isFocusMode();
  setFocusMode(next);
  return next;
}
