🗄 SQL ajouté

-- Nouvelle table pharmacies (référentiel)
CREATE TABLE pharmacies (id, structure_id, nom, finess, adresse, ...)

-- ALTER membres_structure (rôles pro)
ALTER TABLE membres_structure
  ADD COLUMN role_professionnel TEXT,
  ADD COLUMN specialite TEXT,
  ADD COLUMN numero_adeli TEXT, ADD COLUMN numero_rpps TEXT,
  ADD COLUMN pharmacie_id UUID,
  ADD COLUMN etablissement_id UUID, ADD COLUMN service_id UUID,
  ADD COLUMN prenom TEXT, ADD COLUMN nom TEXT,
  ADD COLUMN telephone TEXT;

-- ALTER patients (collaborateur référent)
ALTER TABLE patients ADD COLUMN collaborateur_id UUID;

-- Vue v_collaborateurs (jointures pharmacie/étab/service nom)
CREATE VIEW v_collaborateurs AS SELECT ms.*, ph.nom AS pharmacie_nom, ...
