🗄 SQL créé

-- Table pathologies (référentiel commun structure)
CREATE TABLE pathologies (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  structure_id UUID NOT NULL,
  code TEXT, nom TEXT NOT NULL,
  description TEXT,
  icone TEXT DEFAULT 'ti-stethoscope',
  couleur TEXT DEFAULT '#185FA5',
  protocole_court TEXT,
  protocole_detail TEXT,
  alertes TEXT,
  actif BOOLEAN DEFAULT true,
  ...
);

-- Jointure services <-> pathologies (M:N)
CREATE TABLE services_pathologies (
  service_id UUID NOT NULL,
  pathologie_id UUID NOT NULL,
  PRIMARY KEY (service_id, pathologie_id)
);

-- Lien patient → pathologie principale
ALTER TABLE patients ADD COLUMN pathologie_id UUID;
