💻 Code clé — drag & drop natif

// 0.58.33 - app/accueil/page.js (handlers drag & drop natif)
const [draggedWidget, setDraggedWidget] = useState(null);
const [dragOverWidget, setDragOverWidget] = useState(null);

function handleDragStart(e, k) {
  if (!editLayout) return;
  setDraggedWidget(k);
  e.dataTransfer.effectAllowed = 'move';
}

function handleDragOver(e, k) {
  if (!editLayout || !draggedWidget) return;
  e.preventDefault();  // requis pour autoriser le drop
  e.dataTransfer.dropEffect = 'move';
  if (dragOverWidget !== k && k !== draggedWidget) setDragOverWidget(k);
}

function handleDrop(e, targetK) {
  if (!editLayout || !draggedWidget) return;
  e.preventDefault();
  if (targetK === draggedWidget) {
    setDraggedWidget(null);
    setDragOverWidget(null);
    return;
  }
  const sourceIdx = widgetOrder.indexOf(draggedWidget);
  const targetIdx = widgetOrder.indexOf(targetK);
  const next = [...widgetOrder];
  next.splice(sourceIdx, 1);                  // retire
  next.splice(targetIdx, 0, draggedWidget);   // insère à la position du target
  setWidgetOrder(next);
  setDashboardLayout({ active: widgets, order: next });
  setDraggedWidget(null);
  setDragOverWidget(null);
}

🎁 Ce que tu vas voir

  • /accueil → bouton 'Personnaliser' en haut-droite
  • Clic → MODE ÉDITION s'active avec toolbar premium en haut + bordures dashed sur chaque widget
  • Chaque widget a un label flottant en haut-gauche (icône + drag handle + nom du widget en couleur métier)
  • Chaque widget a une croix rouge en haut-droite (avec animation scale au hover)
  • Glisse un widget vers un autre = reorder fluide avec opacity 0.4 sur le source + translateY sur le target
  • Clic croix = widget masqué + apparaît dans "Widgets masqués" de la toolbar
  • Clic pilule widget masqué = réactivé instantanément
  • Clic 'Réinitialiser' = dialog confirm puis restore des défauts
  • Clic 'Terminé' = ferme l'édition, retour mode lecture
