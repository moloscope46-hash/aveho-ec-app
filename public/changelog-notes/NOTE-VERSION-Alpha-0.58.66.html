🚨 1. FIX build Vercel — stub Supabase SSR-safe

Symptôme : Vercel échouait avec Error: @supabase/ssr: Your project's URL and API key are required to create a Supabase client! au prerender de /materiels et /signalements.

Cause : Next.js 15 tente le SSG → createBrowserClient(undefined, undefined) throw car les NEXT_PUBLIC_* ne sont pas disponibles au build time côté Vercel.

// lib/supabase.js (0.58.66)
export function createClient() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
  if (!url || !key) {
    // 0.58.66 : stub Proxy au lieu de throw (build prerender)
    const asyncStub = () => Promise.resolve({ data: null, error: null });
    const stubBuilder = new Proxy({}, {
      get: (_target, prop) => {
        if (prop === 'then') return undefined;  // pas une promesse
        return () => stubBuilder;
      },
    });
    return {
      from: () => stubBuilder,
      auth: { getUser: asyncStub, getSession: asyncStub, ... },
      storage: { from: () => ({ upload: asyncStub, ... }) },
      rpc: asyncStub, channel: () => ({...}),
      functions: { invoke: asyncStub },
    };
  }
  return createBrowserClient(url, key);
}
