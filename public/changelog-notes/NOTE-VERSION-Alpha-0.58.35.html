🏢 2. TopBar : bâtiment + service courants (desktop only)

Deux sélecteurs compacts ajoutés à côté du sélecteur d'établissement. Persistés en localStorage avec préfixe av- (purgeables au logout). Masqués en mobile via media query (la TopBar mobile a déjà peu de place).

// 0.58.35 - app/components/BatimentServiceSwitcher.js (NEW)
const STORAGE_BAT = 'av-current-batiment-id';
const STORAGE_SVC = 'av-current-service-id';

// Charge bâtiments quand auth.etabId change
useEffect(() => {
  supabase.from('batiments')
    .select('id, nom')
    .eq('etablissement_id', auth.etabId);
}, [auth?.etabId]);

// Charge services via étages → services
useEffect(() => {
  const { data: etages } = await supabase.from('etages')
    .select('id').eq('batiment_id', batId);
  const { data: svcs } = await supabase.from('services')
    .select('id, nom').in('etage_id', etages.map(e => e.id));
}, [batId]);

// Émet event pour que d'autres composants puissent réagir
window.dispatchEvent(new CustomEvent('av-current-context-change', {
  detail: { batimentId: id, serviceId: null }
}));

// CSS : masqué en mobile
@media (max-width: 768px) {
  .bat-svc-switch { display: none !important; }
}
